\section{模型的评价}

\subsection{评价指标}
\begin{itemize}
\item MAE = frac{1}{n} sum_i |hat{y}_i - y_i|
\item RMSE = sqrt(frac{1}{n} sum_i (hat{y}_i - y_i)^2)
\item MAPE = frac{100%}{n} sum_i |frac{hat{y}_i - y_i}{y_i}|
\end{itemize}
